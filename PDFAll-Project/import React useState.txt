import React, { useState, useEffect } from 'react';
import { User, Users, Building2, Plug, FileText, Eye, EyeOff, Edit2, Trash2, X, Loader2 } from 'lucide-react';
import { fetchWithAuth } from '../services/api';

const Parametres = () => {
  const [activeTab, setActiveTab] = useState('api');
  const [showApiKey, setShowApiKey] = useState(false);
  const [isModalOpen, setIsModalOpen] = useState(false);
  const userRole = localStorage.getItem('userRole'); // 'Admin' ou 'User'
  const [team, setTeam] = useState([]);
  const [newStaff, setNewStaff] = useState({ FirstName: '', Email: '', Password: '' });
  // Gestion du Pop-up "Changer le mot de passe"
  const [isPasswordModalOpen, setIsPasswordModalOpen] = useState(false);
  const [passwords, setPasswords] = useState({ old: '', new: '', confirm: '' });
  const [showOldPwd, setShowOldPwd] = useState(false);
  const [showNewPwd, setShowNewPwd] = useState(false);
  const [showConfPwd, setShowConfPwd] = useState(false);


  // États reliés à la base de données
  const [profile, setProfile] = useState({ companyName: '', email: '' });
  const [apiCreds, setApiCreds] = useState({ senderId: '', apiKey: '' });
  const [templates, setTemplates] = useState([]);
  
  const [editingTemplate, setEditingTemplate] = useState(null);
  const [modelName, setModelName] = useState('');
  const [messageContent, setMessageContent] = useState('');
  const [confirmDeleteId, setConfirmDeleteId] = useState(null);
  const [dragOverField, setDragOverField] = useState(false);
  
  // Variables correspondant EXACTEMENT à ce que le moteur Cron sait remplacer !
  const fields = ['Nom', 'Prenom', 'Date', 'Heure'];

  // États pour le test de connexion
  const [connectionStatus, setConnectionStatus] = useState('idle');
  const [connectionMessage, setConnectionMessage] = useState('Cliquez pour tester votre connexion.');
  const [loadingConnection, setLoadingConnection] = useState(false);

  // 1. CHARGEMENT DES DONNÉES AU DÉMARRAGE
  useEffect(() => {
    const loadData = async () => {
      if (userRole === 'Admin') {
          const teamRes = await fetchWithAuth('/api/Users/team');
          if (teamRes.ok) setTeam(await teamRes.json());
        }
      try {
        // Charger le profil (Nom, Email, Clés API)
        const userRes = await fetchWithAuth('/api/Users/profile');
        if (userRes.ok) {
          const userData = await userRes.json();
          setProfile({ companyName: userData.companyName || '', email: userData.email || '' });
          setApiCreds({ senderId: userData.senderId || '', apiKey: userData.apiKey || '' });
        }

        // Charger les Templates
        const tplRes = await fetchWithAuth('/api/SmsTemplates');
        if (tplRes.ok) {
          setTemplates(await tplRes.json());
        }
      } catch (error) {
        console.error("Erreur de chargement des paramètres:", error);
      }
    };
    loadData();
  }, []);

  // 2. SAUVEGARDE DES CLÉS API
  const handleSaveApiCreds = async () => {
    try {
      const res = await fetchWithAuth('/api/Users/ApiCredentials', {
        method: 'PUT',
        body: JSON.stringify(apiCreds)
      });
      if (res.ok) {
        alert("Identifiants API sauvegardés avec succès !");
      }
    } catch (error) {
      alert("Erreur lors de la sauvegarde.");
    }
  };

  // 3. SAUVEGARDE D'UN MODÈLE (Création ou Modification)
  const handleSaveTemplate = async () => {
    if (!modelName || !messageContent) return alert("Veuillez remplir le nom et le contenu.");

    const payload = { templateName: modelName, messageContent: messageContent };
    const isUpdate = editingTemplate != null;
    const url = isUpdate ? `/api/SmsTemplates/${editingTemplate.idTemplate}` : '/api/SmsTemplates';
    
    try {
      const res = await fetchWithAuth(url, {
        method: isUpdate ? 'PUT' : 'POST',
        body: JSON.stringify(payload)
      });

      if (res.ok) {
        // Recharger la liste des templates
        const tplRes = await fetchWithAuth('/api/SmsTemplates');
        if (tplRes.ok) setTemplates(await tplRes.json());
        
        setIsModalOpen(false);
        setEditingTemplate(null);
      }
    } catch (error) {
      alert("Erreur lors de la sauvegarde du modèle.");
    }
  };

  // 5. CHANGEMENT DE MOT DE PASSE
  const handleChangePassword = async (e) => {
    e.preventDefault(); // Empêche la page de se recharger
    
    if (passwords.new !== passwords.confirm) {
      return alert("Les nouveaux mots de passe ne correspondent pas !");
    }
    if (passwords.new.length < 6) {
      return alert("Le nouveau mot de passe doit faire au moins 6 caractères.");
    }

    try {
      const res = await fetchWithAuth('/api/Users/password', {
        method: 'PUT',
        body: JSON.stringify({ oldPassword: passwords.old, newPassword: passwords.new })
      });

      if (res.ok) {
        alert("Mot de passe modifié avec succès !");
        setIsPasswordModalOpen(false);
        setPasswords({ old: '', new: '', confirm: '' }); // On vide les champs
      } else {
        const errorText = await res.text();
        alert(errorText || "Erreur lors de la modification.");
      }
    } catch (error) {
      alert("Erreur réseau de communication.");
    }
  };

  // 4. SUPPRESSION D'UN MODÈLE
  const handleDeleteTemplate = async () => {
    if (!confirmDeleteId) return;
    try {
      const res = await fetchWithAuth(`/api/SmsTemplates/${confirmDeleteId}`, { method: 'DELETE' });
      if (res.ok) {
        setTemplates(prev => prev.filter(tpl => tpl.idTemplate !== confirmDeleteId));
        setConfirmDeleteId(null);
      }
    } catch (error) {
      alert("Erreur de suppression.");
    }
  };

  const testConnection = async () => {
    if (loadingConnection) return;
    
    // Vérification de base
    if (!apiCreds.apiKey || !apiCreds.senderId) {
      setConnectionStatus('error');
      setConnectionMessage("Veuillez remplir les deux champs.");
      return;
    }

    setLoadingConnection(true);
    setConnectionStatus('pending');
    setConnectionMessage('Test de connexion au serveur...');

    try {
      // On interroge notre vrai Backend !
      const res = await fetchWithAuth('/api/Users/test-api', {
        method: 'POST',
        body: JSON.stringify(apiCreds)
      });

      if (res.ok) {
        setConnectionStatus('success');
        setConnectionMessage('Connexion réussie avec DigiSMS !');
      } else {
        const errorText = await res.text();
        setConnectionStatus('error');
        setConnectionMessage(errorText || "Échec de la connexion.");
      }
    } catch (error) {
      setConnectionStatus('error');
      setConnectionMessage("Erreur réseau de communication.");
    } finally {
      setLoadingConnection(false);
    }
  };

  return (
    <div className="flex-1 flex flex-col overflow-y-auto p-10 bg-brand-bg relative">
      <div className="flex gap-10">
        
        {/* ================= MENU GAUCHE (ONGLETS) ================= */}
        <div className="w-1/3 flex flex-col space-y-4">

          {/* Bouton Profil Entreprise */}
          <button onClick={() => setActiveTab('profil')} className={`flex items-center px-6 py-4 rounded-xl font-bold text-lg transition-all ${activeTab === 'profil' ? 'bg-slate-400 text-white shadow-md' : 'bg-slate-300/60 text-slate-800'}`}>
            <Building2 className="w-6 h-6 mr-3" /> Profil de l'Entreprise
          </button>
          
          {/* Les autres boutons (API, Templates) restent identiques... */}
          
          {/* Bouton Mon Équipe */}
          {userRole === 'Admin' && (
            <button onClick={() => setActiveTab('team')} className={`flex items-center px-6 py-4 rounded-xl font-bold text-lg transition-all ${activeTab === 'team' ? 'bg-slate-400 text-white shadow-md' : 'bg-slate-300/60 text-slate-800'}`}>
              <Users className="w-6 h-6 mr-3" /> Mon Équipe
            </button>
          )}
          <button onClick={() => setActiveTab('api')} className={`flex items-center px-6 py-4 rounded-xl font-bold text-lg transition-all ${activeTab === 'api' ? 'bg-slate-400 text-white shadow-md' : 'bg-slate-300/60 text-slate-800'}`}>
            <Plug className="w-6 h-6 mr-3" /> Intégration API (DigiSMS)
          </button>
          <button onClick={() => setActiveTab('templates')} className={`flex items-center px-6 py-4 rounded-xl font-bold text-lg transition-all ${activeTab === 'templates' ? 'bg-slate-400 text-white shadow-md' : 'bg-slate-300/60 text-slate-800'}`}>
            <FileText className="w-6 h-6 mr-3" /> Modèles de SMS (Templates)
          </button>
        </div>

        {/* ================= CONTENU DROITE (CARTE BLANCHE) ================= */}
        <div className="w-2/3 bg-white rounded-[2rem] shadow-lg border border-slate-100 p-10 min-h-[500px]">
          
          {/* ---- ONGLET : PROFIL ---- */}
          {activeTab === 'profil' && (
            <div className="animate-in fade-in duration-300">
              <div className="flex flex-col items-center mb-8">
                
                <Building2 className="w-8 h-8 mr-3" /> Profil de l'Entreprise

                <div className="w-48 h-1 bg-brand-red rounded-full mt-3"></div>
              </div>
              <div className="space-y-6 max-w-md mx-auto">
                <div>
                  <label className="block text-lg font-bold text-brand-dark mb-2">Nom de la Structure</label>
                  <input type="text" value={profile.companyName} readOnly className="w-full px-4 py-3 border border-slate-400 rounded-full text-slate-500 bg-slate-50 outline-none cursor-not-allowed" />
                </div>
                <div>
                  <label className="block text-lg font-bold text-brand-dark mb-2">Adresse E-mail de connexion</label>
                  <input type="email" value={profile.email} readOnly className="w-full px-4 py-3 border border-slate-400 rounded-full text-slate-500 bg-slate-50 outline-none cursor-not-allowed" />
                </div>
                <div className="flex justify-center pt-8">
                  <button onClick={() => setIsPasswordModalOpen(true)} className="px-8 py-3 bg-white text-brand-dark font-extrabold text-xl rounded-2xl border-b-4 border-brand-red shadow-sm active:translate-y-[1px] transition-transform">
                    Modifier le mot de passe
                  </button>
                </div>
              </div>
            </div>
          )}

          {/* ---- ONGLET : API DIGISMS ---- */}
          {activeTab === 'api' && (
            <div className="animate-in fade-in duration-300">
              <div className="flex flex-col items-center mb-8">
                <h2 className="text-3xl font-extrabold text-brand-dark flex items-center">
                  <Plug className="w-8 h-8 mr-3" /> Intégration API (DigiSMS)
                </h2>
                <div className="w-48 h-1 bg-brand-red rounded-full mt-3"></div>
              </div>
              <div className="space-y-6 max-w-md mx-auto">
                <div>
                  <label className="block text-lg font-bold text-brand-dark mb-2">Nom d'Expéditeur (Sender ID)</label>
                  <input 
                    type="text" 
                    value={apiCreds.senderId} 
                    onChange={e => setApiCreds({...apiCreds, senderId: e.target.value})} 
                    placeholder="DIGIALERT" 
                    className="w-full px-4 py-3 border border-slate-400 rounded-full text-brand-dark focus:ring-2 focus:ring-brand-red/20 outline-none uppercase" 
                  />
                </div>
                <div>
                  <label className="block text-lg font-bold text-brand-dark mb-2">Clé d'API (API Key)</label>
                  <div className="relative">
                    <input 
                      type={showApiKey ? "text" : "password"} 
                      value={apiCreds.apiKey} 
                      onChange={e => setApiCreds({...apiCreds, apiKey: e.target.value})} 
                      placeholder="Collez votre clé API ici..." 
                      className="w-full pl-4 pr-12 py-3 border border-slate-400 rounded-full text-brand-dark focus:ring-2 focus:ring-brand-red/20 outline-none tracking-wider" 
                    />
                    <button onClick={() => setShowApiKey(!showApiKey)} className="absolute right-4 top-1/2 transform -translate-y-1/2 text-slate-500 active:text-brand-dark">
                      {showApiKey ? <EyeOff className="w-6 h-6" /> : <Eye className="w-6 h-6" />}
                    </button>
                  </div>
                </div>
                
                <div className="flex flex-col items-center pt-6">
                  <button type="button" onClick={testConnection} className="flex items-center justify-center space-x-3 px-6 py-3 bg-white text-brand-dark font-extrabold text-xl rounded-2xl border-b-4 border-brand-red shadow-sm transition-all active:translate-y-[1px]">
                    {loadingConnection ? <Loader2 className="w-5 h-5 animate-spin text-brand-dark" /> : null}
                    <span>Tester la connexion</span>
                    <span className={`w-6 h-6 rounded-full ${connectionStatus === 'success' ? 'bg-green-500 border-2 border-green-700 shadow-[0_0_8px_rgba(34,197,94,0.6)]' : connectionStatus === 'error' ? 'bg-red-500 border-2 border-red-700 shadow-[0_0_8px_rgba(239,68,68,0.4)]' : 'bg-slate-300 border border-slate-400'}`}></span>
                  </button>
                  <p className={`text-sm font-bold text-center mt-3 ${connectionStatus === 'error' ? 'text-red-600' : connectionStatus === 'success' ? 'text-emerald-600' : 'text-slate-600'}`}>{connectionMessage}</p>
                </div>

                <div className="flex justify-center pt-8">
                  <button onClick={handleSaveApiCreds} className="px-8 py-3 bg-brand-dark text-white font-extrabold text-xl rounded-2xl border-b-4 border-slate-800 shadow-sm active:translate-y-[1px] transition-transform">
                    Sauvegarder la clé API
                  </button>
                </div>
              </div>
            </div>
          )}

          {/* ---- ONGLET : ÉQUIPE ---- */}
          {activeTab === 'team' && userRole === 'Admin' && (
            <div className="animate-in fade-in duration-300">
              <Users className="w-8 h-8 mr-3" /> Gérer l'Équipe
              
              <div className="mb-10 bg-slate-50 p-6 rounded-2xl border border-slate-200">
                <h3 className="font-bold text-brand-dark mb-4">Ajouter un collaborateur</h3>
                <div className="flex gap-4">
                  <input type="text" placeholder="Prénom" value={newStaff.FirstName} onChange={e => setNewStaff({...newStaff, FirstName: e.target.value})} className="flex-1 px-4 py-2 rounded-lg border outline-none focus:border-brand-red" />
                  <input type="email" placeholder="Email" value={newStaff.Email} onChange={e => setNewStaff({...newStaff, Email: e.target.value})} className="flex-1 px-4 py-2 rounded-lg border outline-none focus:border-brand-red" />
                  <input type="password" placeholder="Mot de passe" value={newStaff.Password} onChange={e => setNewStaff({...newStaff, Password: e.target.value})} className="flex-1 px-4 py-2 rounded-lg border outline-none focus:border-brand-red" />
                  <button onClick={async () => {
                    const res = await fetchWithAuth('/api/Users/team', { method: 'POST', body: JSON.stringify(newStaff) });
                    if (res.ok) {
                      alert("Employé ajouté !");
                      setNewStaff({ FirstName: '', Email: '', Password: '' });
                      const teamRes = await fetchWithAuth('/api/Users/team');
                      if (teamRes.ok) setTeam(await teamRes.json());
                    }
                  }} className="px-6 py-2 bg-brand-red text-white font-bold rounded-lg hover:bg-rose-700">Ajouter</button>
                </div>
              </div>

              <table className="w-full text-left">
                <thead><tr className="border-b text-slate-500"><th className="pb-3">Prénom / Clinique</th><th className="pb-3">Email</th><th className="pb-3">Rôle</th></tr></thead>
                <tbody>
                  {team.map(member => (
                    <tr key={member.idUser} className="border-b">
                      <td className="py-4 font-bold text-brand-dark">{member.companyName}</td>
                      <td className="py-4 text-slate-600">{member.email}</td>
                      <td className="py-4"><span className={`px-3 py-1 rounded-full text-xs font-bold ${member.role === 'Admin' ? 'bg-brand-dark text-white' : 'bg-slate-200 text-slate-700'}`}>{member.role}</span></td>
                    </tr>
                  ))}
                </tbody>
              </table>
            </div>
          )}

          {/* ---- ONGLET : TEMPLATES ---- */}
          {activeTab === 'templates' && (
            <div className="animate-in fade-in duration-300">
              <div className="flex flex-col items-center mb-8">
                <h2 className="text-3xl font-extrabold text-brand-dark flex items-center">
                  <FileText className="w-8 h-8 mr-3" /> Modèles de SMS (Templates)
                </h2>
                <div className="w-48 h-1 bg-brand-red rounded-full mt-3"></div>
              </div>

              <div className="max-w-lg mx-auto relative border-2 border-brand-dark rounded-3xl p-6 mt-6">
                <div className="absolute -top-4 left-1/2 transform -translate-x-1/2 bg-brand-dark text-white px-6 py-1 rounded-full font-bold text-lg">
                  Vos Messages
                </div>
                
                <div className="space-y-4 mt-2 max-h-[300px] overflow-y-auto pr-2">
                  {templates.map((tpl) => (
                    <div key={tpl.idTemplate} className="flex justify-between items-center border-2 border-brand-red rounded-full px-6 py-3 bg-white shadow-sm">
                      <span className="font-extrabold text-lg text-brand-dark truncate max-w-[200px]" title={tpl.templateName}>{tpl.templateName}</span>
                      <div className="flex space-x-3 shrink-0">
                        <button onClick={() => { setEditingTemplate(tpl); setModelName(tpl.templateName); setMessageContent(tpl.messageContent); setIsModalOpen(true); setConfirmDeleteId(null); }} className="w-8 h-8 rounded-full bg-slate-100 flex items-center justify-center text-brand-dark active:scale-95 transition-transform">
                          <Edit2 className="w-4 h-4" />
                        </button>
                        <button onClick={() => setConfirmDeleteId(tpl.idTemplate)} className="w-8 h-8 rounded-full bg-slate-100 flex items-center justify-center text-brand-red active:scale-95 transition-transform">
                          <Trash2 className="w-4 h-4" />
                        </button>
                      </div>
                    </div>
                  ))}
                  {templates.length === 0 && <p className="text-center text-slate-500 py-4">Aucun modèle créé.</p>}
                </div>
                {confirmDeleteId && (
                  <div className="mt-4 rounded-2xl border border-red-200 bg-red-50 p-4">
                    <p className="text-sm font-bold text-red-700">Voulez-vous vraiment supprimer ce modèle ?</p>
                    <div className="mt-3 flex items-center gap-3">
                      <button onClick={handleDeleteTemplate} className="px-4 py-2 bg-red-600 text-white rounded-full font-bold active:translate-y-[1px] transition-all">Supprimer</button>
                      <button onClick={() => setConfirmDeleteId(null)} className="px-4 py-2 bg-white text-brand-dark border border-slate-300 rounded-full font-bold active:translate-y-[1px] transition-all">Annuler</button>
                    </div>
                  </div>
                )}
              </div>

              <div className="flex justify-center pt-10">
                <button 
                  onClick={() => {
                    setEditingTemplate(null);
                    setModelName('');
                    setMessageContent('');
                    setIsModalOpen(true);
                  }}
                  className="px-8 py-3 bg-brand-dark text-white font-extrabold text-xl rounded-2xl shadow-sm hover:bg-slate-800 active:scale-95 transition-all"
                >
                  Nouveau Modèle
                </button>
              </div>
            </div>
          )}
        </div>
      </div>

      {/* ================= MODAL : NOUVEAU MODÈLE ================= */}
      {isModalOpen && (
        <div className="absolute inset-0 bg-brand-dark/40 backdrop-blur-sm flex items-center justify-center z-50">
          <div className="bg-white rounded-[2.5rem] shadow-2xl w-full max-w-2xl p-10 relative animate-in zoom-in duration-200">

            <div className="flex flex-col items-center mb-8 relative">
              <button onClick={() => setIsModalOpen(false)} className="absolute right-0 top-0 text-slate-500 hover:text-brand-red">
                <X className="w-8 h-8" />
              </button>
              <h2 className="text-4xl font-extrabold text-brand-dark mb-2">
                {editingTemplate ? 'Modifier le Modèle' : 'Nouveau Modèle'}
              </h2>
              <div className="w-40 h-1.5 bg-brand-red rounded-full"></div>
            </div>

            <div className="mb-8">
              <label className="block text-xl font-bold text-brand-dark mb-2">Nom du Modèle</label>
              <input type="text" value={modelName} onChange={e => setModelName(e.target.value)} placeholder="Ex: Rappel 24h avant" className="w-full px-4 py-3 border border-slate-400 rounded-full text-slate-600 focus:ring-2 focus:ring-brand-red/20 outline-none" />
            </div>

            <div className="flex gap-6 mb-8">
              <div className={`flex-1 relative border-[3px] border-brand-dark rounded-2xl p-6 mt-4 ${dragOverField ? 'bg-slate-50 border-dashed border-brand-red' : ''}`} onDragOver={e => { e.preventDefault(); setDragOverField(true); }} onDragLeave={() => setDragOverField(false)} onDrop={e => { e.preventDefault(); const field = e.dataTransfer.getData('text/plain'); if (field) setMessageContent(prev => `${prev}${prev && !prev.endsWith(' ') ? ' ' : ''}{${field}}`); setDragOverField(false); }}>
                <div className="absolute -top-4 left-6 bg-white px-2 font-extrabold text-lg text-brand-dark">
                  Contenu du message
                </div>
                <div className="text-slate-500 text-sm italic mb-2">
                  Glissez-déposez les champs ici ou tapez votre texte.
                </div>
                <textarea value={messageContent} onChange={e => setMessageContent(e.target.value)} className="w-full h-32 mt-2 resize-none outline-none text-xl font-medium text-slate-600" placeholder="Bonjour {Prenom}, votre rendez-vous est prévu le {Date} à {Heure}"></textarea>
              </div>

              <div className="w-1/3 relative border-[3px] border-brand-dark rounded-2xl p-6 mt-4">
                <div className="absolute -top-4 left-6 bg-white px-2 font-extrabold text-lg text-brand-dark">
                  Champs
                </div>
                <div className="flex flex-col space-y-3 mt-2">
                  {fields.map((field) => (
                    <div key={field} className="bg-rose-200 border border-brand-red text-slate-800 px-4 py-1.5 rounded-full font-bold text-sm w-max cursor-move hover:scale-105 transition-transform" draggable onDragStart={e => e.dataTransfer.setData('text/plain', field)}>
                      {field}
                    </div>
                  ))}
                </div>
              </div>
            </div>

            <div className="flex justify-center gap-4 pt-2">
              <button onClick={() => { setIsModalOpen(false); setEditingTemplate(null); }} className="px-10 py-3 bg-white text-slate-500 font-extrabold text-xl rounded-2xl border-2 border-slate-200 hover:bg-slate-50 transition-all">
                Annuler
              </button>
              <button onClick={handleSaveTemplate} className="px-10 py-3 bg-brand-red text-white font-extrabold text-xl rounded-2xl shadow-sm hover:bg-rose-700 active:scale-95 transition-all">
                {editingTemplate ? 'Mettre à jour' : 'Créer le Modèle'}
              </button>
            </div>
          </div>
        </div>
      )}
      {/* ================= MODAL : CHANGER LE MOT DE PASSE ================= */}
      {isPasswordModalOpen && (
        <div className="absolute inset-0 bg-brand-dark/40 backdrop-blur-sm flex items-center justify-center z-50">
          <div className="bg-white rounded-[2.5rem] shadow-2xl w-full max-w-md p-10 relative animate-in zoom-in duration-200">

            <div className="flex flex-col items-center mb-8 relative">
              <button onClick={() => setIsPasswordModalOpen(false)} className="absolute right-0 top-0 text-slate-500 hover:text-brand-red">
                <X className="w-8 h-8" />
              </button>
              <h2 className="text-3xl font-extrabold text-brand-dark mb-2 text-center">
                Sécurité
              </h2>
              <div className="w-32 h-1.5 bg-brand-red rounded-full"></div>
            </div>

            <form onSubmit={handleChangePassword} className="space-y-5">
              <div>
                <label className="block text-sm font-bold text-brand-dark mb-2">Ancien mot de passe</label>
                <div className="relative">
                  <input 
                    type={showOldPwd ? "text" : "password"} 
                    required
                    value={passwords.old} 
                    onChange={e => setPasswords({...passwords, old: e.target.value})} 
                    className="w-full pl-4 pr-12 py-3 border border-slate-300 rounded-xl text-brand-dark focus:ring-2 focus:ring-brand-red/20 outline-none [&::-ms-reveal]:hidden" 
                  />
                  <button type="button" onClick={() => setShowOldPwd(!showOldPwd)} className="absolute right-4 top-1/2 transform -translate-y-1/2 text-slate-400 hover:text-brand-dark transition-colors">
                    {showOldPwd ? <EyeOff className="w-5 h-5" /> : <Eye className="w-5 h-5" />}
                  </button>
                </div>
              </div>

              <div>
                <label className="block text-sm font-bold text-brand-dark mb-2">Nouveau mot de passe</label>
                <div className="relative">
                  <input 
                    type={showNewPwd ? "text" : "password"} 
                    required
                    value={passwords.new} 
                    onChange={e => setPasswords({...passwords, new: e.target.value})} 
                    className="w-full pl-4 pr-12 py-3 border border-slate-300 rounded-xl text-brand-dark focus:ring-2 focus:ring-brand-red/20 outline-none [&::-ms-reveal]:hidden" 
                  />
                  <button type="button" onClick={() => setShowNewPwd(!showNewPwd)} className="absolute right-4 top-1/2 transform -translate-y-1/2 text-slate-400 hover:text-brand-dark transition-colors">
                    {showNewPwd ? <EyeOff className="w-5 h-5" /> : <Eye className="w-5 h-5" />}
                  </button>
                </div>
              </div>

              <div>
                <label className="block text-sm font-bold text-brand-dark mb-2">Confirmer le nouveau mot de passe</label>
                <div className="relative">
                  <input 
                    type={showConfPwd ? "text" : "password"} 
                    required
                    value={passwords.confirm} 
                    onChange={e => setPasswords({...passwords, confirm: e.target.value})} 
                    className="w-full pl-4 pr-12 py-3 border border-slate-300 rounded-xl text-brand-dark focus:ring-2 focus:ring-brand-red/20 outline-none [&::-ms-reveal]:hidden" 
                  />
                  <button type="button" onClick={() => setShowConfPwd(!showConfPwd)} className="absolute right-4 top-1/2 transform -translate-y-1/2 text-slate-400 hover:text-brand-dark transition-colors">
                    {showConfPwd ? <EyeOff className="w-5 h-5" /> : <Eye className="w-5 h-5" />}
                  </button>
                </div>
              </div>

              <div className="flex justify-center pt-6">
                <button type="submit" className="w-full py-3 bg-brand-red text-white font-extrabold text-lg rounded-xl shadow-sm hover:bg-rose-700 active:scale-95 transition-all">
                  Mettre à jour
                </button>
              </div>
            </form>
            
          </div>
        </div>
      )}
    </div>
  );
};

export default Parametres;
